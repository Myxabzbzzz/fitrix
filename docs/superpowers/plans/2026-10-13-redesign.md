# FITRIX Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bring the Flutter app in line with the current FITRIX design (prototype + Figma tokens), add the missing screens (Discover, Shop, Fitrix map with travel modes) and the Supabase backend they need, with tests, ready to run on an iPhone.

**Architecture:** One token layer (`AppPalette` + new `AppSpacing`/`AppRadius`/`AppText`) feeds a small set of shared widgets that mirror the Figma components; screens are rebuilt from those widgets. New content (shop catalog, Discover, places) lives in Supabase tables with public read and RLS-protected user rows; the app reads it through repositories with a bundled offline fallback. The map uses `flutter_map` with OSM tiles and a pure-Dart router (A* walk/bike/car + metro planner) over the bundled graph from the prototype.

**Tech Stack:** Flutter 3 (Riverpod 2, go_router 14, dio, supabase_flutter 2), flutter_map, geolocator, Supabase Postgres (migrations + pgTAP), Node backend (unchanged except tests).

**Spec:** the design source of truth is the prototype `design/prototype/index.html` (published: https://claude.ai/artifact/1UTp3z1bryWXXonFpmHJ2K) and the Figma file `w0TugufzKexAAU34FFk6W8` (variables `Color`, `Dimensions`, text styles). Decisions from the owner (2026-10-13): iPhone build; flutter_map + OSM tiles (replace tiles with a keyed provider before release); new Supabase tables, verified locally then applied to the cloud project only after explicit owner approval; no payments (orders are saved as requests; FITRIX Pro is a screen only).

```sql
create table public.products (
  id text primary key check (char_length(id) <= 60),
  name text not null check (char_length(name) between 1 and 100),
  category text not null check (category in ('supplements', 'equipment', 'apparel')),
  price_cents integer not null check (price_cents between 0 and 10000000),
  currency text not null default 'USD' check (currency in ('USD', 'UZS')),
  image text check (char_length(image) <= 100),       -- asset name, e.g. 'tile_workouts'
  word text check (char_length(word) <= 12),           -- typographic placeholder when no image
  position integer not null default 0,
  active boolean not null default true
);
create table public.orders (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users (id) on delete cascade,
  status text not null default 'placed' check (status in ('placed', 'cancelled', 'fulfilled')),
  total_cents integer not null check (total_cents >= 0),
  currency text not null,
  created_at timestamptz not null default now()
);
create table public.order_items (
  order_id uuid not null references public.orders (id) on delete cascade,
  product_id text not null references public.products (id),
  quantity integer not null check (quantity between 1 and 20),
  price_cents integer not null check (price_cents >= 0),
  primary key (order_id, product_id)
);
create table public.programs (id text primary key, title text not null, meta text not null, level text not null check (level in ('beginner','medium','advanced')), sport text not null, image text, position integer not null default 0);
create table public.articles (id text primary key, tag text not null, title text not null, minutes smallint not null check (minutes between 1 and 60), body text not null, position integer not null default 0);
create table public.challenges (id text primary key, title text not null, description text not null, days smallint not null check (days between 1 and 60), daily_target text not null, starts_on date not null, ends_on date not null, check (ends_on >= starts_on));
create table public.challenge_participants (
  challenge_id text not null references public.challenges (id) on delete cascade,
  user_id uuid not null references auth.users (id) on delete cascade,
  joined_at timestamptz not null default now(),
  primary key (challenge_id, user_id)
);
create table public.places (id text primary key, name text not null, kind text not null check (kind in ('gyms','parks','tracks','pools')), type text not null, lat double precision not null, lon double precision not null, hours text, note text, osm_ref text);

-- Orders are only created through this function so prices come from the catalog, never from the client.
create function public.place_order(p_items jsonb) returns uuid
language plpgsql security definer set search_path = public as $$
declare v_order uuid; v_total integer := 0; v_currency text; r record;
begin
  if auth.uid() is null then raise exception 'not signed in' using errcode = '28000'; end if;
  if jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 or jsonb_array_length(p_items) > 30 then
    raise exception 'invalid items' using errcode = '22023'; end if;
  insert into orders (user_id, total_cents, currency) values (auth.uid(), 0, 'USD') returning id into v_order;
  for r in select (e->>'product_id') as pid, (e->>'quantity')::int as qty from jsonb_array_elements(p_items) e loop
    insert into order_items (order_id, product_id, quantity, price_cents)
      select v_order, p.id, r.qty, p.price_cents from products p where p.id = r.pid and p.active;
    if not found then raise exception 'unknown product %', r.pid using errcode = '22023'; end if;
  end loop;
  select sum(quantity * price_cents) into v_total from order_items where order_id = v_order;
  update orders set total_cents = v_total where id = v_order;
  return v_order;
end $$;
revoke all on function public.place_order(jsonb) from public, anon;
grant execute on function public.place_order(jsonb) to authenticated;
```
